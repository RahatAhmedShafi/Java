public class Point {
    int x, y;
    Point(int x, int y){
        this.x = x;
        this.y = y;
    }
    void right(){
        this.x = this.x + 1;
    }
    void left(){
        this.x = this.x - 1;
    }
    void up(){
        this.y = this.y + 1;
    }
    void down(){
        this.y = this.y - 1;
    }
    void show(){
        System.out.println("x = " + this.x + " y = " + this.y);
//        System.out.println(this.y);
    }
    public static void main(String[] args) {
        Point p1 = new Point(1, 2);
        Point p2 = new Point(5, 6);
        p1.show();
        p1.right(); // 2 2
        p1.up(); // 2 3
        p1.left(); // 1 3
        p1.show();
        p2.right(); // 6 6
        p2.down(); // 6 5
        p2.show();
    }
}
