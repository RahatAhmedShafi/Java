class Asset{
    String assetType;
    private float price;
    int id;
    String tradersName;
    static int id_counter = 1001;
    Asset(String assetType, float price){
        if(assetType == "Bond" || assetType == "Stock")
            this.assetType = assetType;
        setPrice(price);
        id = id_counter;
        id_counter++;
    }
    public void setPrice(float price){
        if(price >= 0)
            this.price = price;
    }
}


class Trader{
    String name;
    String department = "Bond";
    float aum = 0.0f;
    Trader(String name, String department){
        this.name = name;
        this.department = department;
    }
    Trader(String name){
        this.name = name;
    }
    public void assign(Asset asset){
        if(asset.assetType == this.department){
            aum += asset.price;
        } else{
            sout("Department Mismatch");
        }
    }
}
